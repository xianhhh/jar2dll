cmake_minimum_required(VERSION 3.21)
project(jar2dll_jni_fixtures LANGUAGES CXX)

# These fixtures intentionally remain separate DLLs. They are the input corpus for
# the experimental memory loader; do not merge them into maindll.
find_package(JNI REQUIRED)

add_library(j2d_basic_native SHARED basic_jni.cpp)
target_include_directories(j2d_basic_native PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_features(j2d_basic_native PRIVATE cxx_std_20)
set_target_properties(j2d_basic_native PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF)

add_library(j2d_overload_native SHARED overload_jni.cpp)
target_include_directories(j2d_overload_native PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_features(j2d_overload_native PRIVATE cxx_std_20)
set_target_properties(j2d_overload_native PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF)

add_library(j2d_complex_native SHARED complex_jni.cpp)
target_include_directories(j2d_complex_native PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_features(j2d_complex_native PRIVATE cxx_std_20)
set_target_properties(j2d_complex_native PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF)

add_library(j2d_fixture_provider SHARED dependency_provider.cpp)
add_library(j2d_dependent_native SHARED dependent_jni.cpp)
target_include_directories(j2d_dependent_native PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(j2d_dependent_native PRIVATE j2d_fixture_provider)
target_compile_features(j2d_dependent_native PRIVATE cxx_std_20)
set_target_properties(j2d_dependent_native PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF)

add_executable(j2d_memory_pe_test memory_pe_tests.cpp)
target_compile_features(j2d_memory_pe_test PRIVATE cxx_std_20)
add_dependencies(j2d_memory_pe_test j2d_basic_native)

add_library(j2d_chain_provider SHARED chain_provider.cpp)
add_library(j2d_chain_bridge SHARED chain_bridge.cpp)
target_link_libraries(j2d_chain_bridge PRIVATE j2d_chain_provider)
add_library(j2d_chain_jni SHARED chain_jni.cpp)
target_include_directories(j2d_chain_jni PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(j2d_chain_jni PRIVATE j2d_chain_bridge)

add_library(j2d_lifecycle_fixture SHARED lifecycle_fixture.cpp)
target_link_options(j2d_lifecycle_fixture PRIVATE /ENTRY:FixtureEntry /NODEFAULTLIB)
add_executable(j2d_memory_lifecycle_test memory_lifecycle_tests.cpp)
add_dependencies(j2d_memory_lifecycle_test j2d_lifecycle_fixture)
add_executable(j2d_memory_negative_test memory_negative_tests.cpp)
add_executable(j2d_memory_registry_negative_test memory_registry_negative_tests.cpp)
add_executable(j2d_memory_import_test memory_import_tests.cpp)
add_executable(j2d_memory_registry_test memory_registry_tests.cpp)
add_executable(j2d_jni_exports_test jni_exports_tests.cpp)
add_executable(j2d_memory_bundle_test memory_bundle_tests.cpp)

include(CTest)
if(BUILD_TESTING)
    add_test(NAME memory_pe_map_and_imports COMMAND j2d_memory_pe_test $<TARGET_FILE:j2d_basic_native>)
    add_test(NAME memory_pe_chain_preflight COMMAND j2d_memory_pe_test $<TARGET_FILE:j2d_chain_jni>)
    add_test(NAME memory_pe_provider_export COMMAND j2d_memory_pe_test $<TARGET_FILE:j2d_fixture_provider> j2d_fixture_scale)
    add_test(NAME memory_internal_import_stress COMMAND j2d_memory_import_test $<TARGET_FILE:j2d_fixture_provider> $<TARGET_FILE:j2d_dependent_native> 1000)
    add_test(NAME memory_registry_chain COMMAND j2d_memory_registry_test $<TARGET_FILE:j2d_chain_provider> $<TARGET_FILE:j2d_chain_bridge> $<TARGET_FILE:j2d_chain_jni>)
    add_test(NAME memory_lifecycle_stress COMMAND j2d_memory_lifecycle_test $<TARGET_FILE:j2d_lifecycle_fixture> 1000)
    add_test(NAME memory_negative_input COMMAND j2d_memory_negative_test $<TARGET_FILE:j2d_basic_native>)
    add_test(NAME memory_registry_negative COMMAND j2d_memory_registry_negative_test $<TARGET_FILE:j2d_fixture_provider> $<TARGET_FILE:j2d_dependent_native>)
    add_test(NAME jni_export_names COMMAND j2d_jni_exports_test)
    add_test(NAME memory_bundle_format COMMAND j2d_memory_bundle_test)
endif()
